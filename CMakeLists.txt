cmake_minimum_required(VERSION 3.12)
project(terra_fovea CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenMP)
add_library(terra_fovea SHARED core/terra_fovea.cpp)
add_executable(terra_cli core/terra_cli.cpp core/terra_fovea.cpp)
if(OpenMP_CXX_FOUND)
  target_link_libraries(terra_fovea PRIVATE OpenMP::OpenMP_CXX)
  target_link_libraries(terra_cli PRIVATE OpenMP::OpenMP_CXX)
endif()
set_target_properties(terra_fovea terra_cli PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/build RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/build)
